pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps { git ' }
        }

        stage('Test & Build') {
            steps {
                sh './test.sh'
                sh 'docker build -t myexercise .'
            }
        }

        stage('Deploy') {
            steps {
                sh 'docker rm -f myapp || true'
                sh 'docker run -d --name myexercise -p 8080:8080 myexercise'
            }
        }

        stage('Verify') {
            steps {
                sh 'sleep 5 && curl -f http://localhost:8080'
            }
        }
    }

    post {
        always {
            sh 'docker rm -f my exercise || true'
        }
    }
}
